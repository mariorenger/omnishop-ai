\section{Bài toán và dữ liệu}
%% ============================================================================ Bài toán
\begin{frame}[shrink]{Bài toán khuyến nghị tin tức}
\begin{columns}[T,onlytextwidth]
\column{0.56\textwidth}
\footnotesize
Mỗi \textbf{lượt hiển thị} (impression) gồm lịch sử đọc $\mathcal H_u=[d_1,\dots,d_L]$ của độc giả $u$ và danh sách ứng viên $\mathcal C_u$.
Mô hình chấm điểm $s_\theta(u,c)$ cho từng ứng viên để xếp hạng; nhãn là bài nào được click. Ba đặc điểm làm bài toán khó:
\begin{itemize}
  \setlength{\itemsep}{1pt}
  \item \textbf{Bài mới chiếm đa số (cold-start):} phần lớn click rơi vào bài chưa có trong dữ liệu huấn luyện, nên lọc cộng tác theo ID hoạt động kém.
  \item \textbf{Giá trị tin giảm nhanh theo thời gian:} phân phối dữ liệu dịch chuyển giữa quá khứ (huấn luyện) và tương lai (test).
  \item \textbf{Ràng buộc độ trễ khi phục vụ:} không thể chạy LLM nhiều tỷ tham số cho từng lượt hiển thị.
\end{itemize}
\ychinh{cần biểu diễn \emph{nội dung} bài báo đủ tốt để xếp hạng cả bài chưa từng thấy, với chi phí phục vụ thấp.}
\column{0.41\textwidth}
\centering
\begin{tikzpicture}[font=\scriptsize,>=Stealth,
  doc/.style={draw=keyc,fill=keyc!6,minimum width=0.62cm,minimum height=0.5cm,inner sep=1pt},
  cand/.style={draw=accentc,fill=accentc!5,minimum width=0.62cm,minimum height=0.5cm,inner sep=1pt}]
\node[anchor=west,font=\scriptsize\bfseries,text=keyc] at (-0.1,2.75) {Lịch sử đọc $\mathcal H_u$};
\foreach \i/\lab in {0/$d_1$,1/$d_2$,2/$d_3$,3/$\cdots$,4/$d_L$} \node[doc] at (0.35+\i*0.72,2.25) {\lab};
\node[anchor=west,font=\scriptsize\bfseries,text=accentc] at (-0.1,1.45) {Ứng viên $\mathcal C_u$};
\foreach \i/\lab in {0/$c_1$,1/$c_2$,2/$c_3$,3/$c_4$,4/$c_5$} \node[cand] (c\i) at (0.35+\i*0.72,0.95) {\lab};
\node[below=1pt of c0,font=\tiny,text=okc] {click};
\node[below=1pt of c3,font=\tiny,text=okc] {click};

\end{tikzpicture}
\end{columns}
\par\kern0pt% chiều cao khung tính cả phần sâu của hộp cuối
\note{Một lượt hiển thị gồm lịch sử đọc của người dùng và danh sách bài được hiển thị; nhiệm vụ là xếp các bài được click lên đầu. Khó vì đa số bài là bài mới, sở thích trôi theo thời gian, và phải phục vụ nhanh. Vì vậy hướng đi của nhóm là dùng LLM để hiểu nội dung bài, nhưng giữ chi phí lúc phục vụ thấp.}
\end{frame}

\ghichu{Test = tập \code{*\_dev} (nhãn của test chính thức không công khai), chỉ chạm một lần ở cuối. Validation là ngày cuối của tập huấn luyện.}
\column{0.47\textwidth}
\begin{block}{Cách đọc các chỉ số}
\scriptsize
\begin{itemize}
  \setlength{\itemsep}{2pt}
  \item \textbf{AUC}: xác suất một bài được click xếp trên một bài không được click trong cùng lượt hiển thị (0.5 là ngẫu nhiên).
  \item \textbf{MRR}: trung bình của 1\,/\,thứ hạng các bài được click.
  \item \textbf{nDCG@10} (chỉ số chính): chất lượng 10 vị trí đầu; bài được click càng lên cao, điểm càng lớn.
  \item \textbf{Cold AUC / Warm AUC}: AUC chỉ tính trên các bài được click \emph{chưa từng} / \emph{đã từng} xuất hiện khi huấn luyện.
\end{itemize}
\end{block}
\end{columns}

\note{MIND-small dùng để sàng lọc nhanh nhiều hướng; MIND-large dùng để xác nhận và xây hệ cuối. Chỉ số chính là nDCG@10. Cold AUC rất quan trọng vì hơn 80\% click là bài mới: một kỹ thuật làm tăng nDCG nhưng giảm Cold AUC thường chỉ đang học thuộc bài cũ.}
\end{frame}

%% ============================================================================ Giao thức
\begin{frame}[shrink]{Giao thức đánh giá và cách đọc một kết quả}
\begin{columns}[T,onlytextwidth]
\column{0.52\textwidth}
\scriptsize
\begin{itemize}
  \setlength{\itemsep}{3pt}
  \item \textbf{Chia theo thời gian:} validation là ngày cuối của tập huấn luyện; test (\code{*\_dev}) chỉ chạm một lần ở cuối. Mọi lựa chọn (tham số, epoch, trọng số ghép, cấu hình stacker) chỉ dùng validation.
  \item \textbf{So sánh ghép cặp theo từng impression:} báo $\Delta$ và khoảng tin cậy (CI) 95\% xấp xỉ chuẩn, $\pm1.96\cdot$SE; Cold/Warm AUC dùng SE theo cụm impression.
  \item \textbf{Chọn đại diện của $m$ họ trên validation}, kết luận có hiệu chỉnh Bonferroni: $z_{\mathrm{adj}}=\Phi^{-1}\!\left(1-\frac{0.05}{2m}\right)$.
  \item Báo cáo riêng \textbf{Cold AUC / Warm AUC}; mô hình có huấn luyện chạy 2--3 seed.
  \item \textbf{73 kiểm thử tự động} trước mỗi lần chạy (đúng chỉ số, rò rỉ nhãn, bộ nhớ); dự đoán kết quả được ghi lại trước khi chạy.
\end{itemize}
\column{0.45\textwidth}
\begin{exampleblock}{Cách đọc ``$\Delta$ [CI 95\%]''}
\scriptsize
BGE-large so với Qwen3 (cùng reader):\\
\quad $+0.0102\ [+0.0095, +0.0108]$ \;$\Rightarrow$ CI không chứa 0: \textbf{khác biệt có ý nghĩa}.\par\vspace{3pt}
kNN trong reader, trên vector nối:\\
\quad $+0.0003\ [-0.0001, +0.0007]$ \;$\Rightarrow$ CI chứa 0: \textbf{coi như không khác}.
\end{exampleblock}
\begin{alertblock}{Lưu ý: nhiễu seed}
\scriptsize CI chỉ tính nhiễu của mẫu test. Hai seed của cùng một mô hình lệch \textbf{0.007--0.014} nDCG@10 trên MIND-small, nên chênh nhỏ hơn mức này giữa hai mô hình đã huấn luyện chưa phải kết luận.
\end{alertblock}
\end{columns}
\par\kern0pt% chiều cao khung tính cả phần sâu của hộp cuối
\note{Giao thức này là lý do nhiều hướng ``trông tốt'' trong văn liệu lại không đứng vững ở đây. Cách đọc đơn giản: nếu khoảng tin cậy không chứa 0 thì khác biệt có ý nghĩa thống kê. Nhưng khoảng tin cậy không tính nhiễu do seed huấn luyện, nên với mô hình đã huấn luyện cần thêm biên an toàn.}
\end{frame}

\section{Các hướng thành công}
%% ============================================================================ H8 trực giác
\begin{frame}[shrink]{H8 \textperiodcentered{} Vì sao mean pooling làm loãng sở thích ngách}
\scriptsize
\[
s_{\mathrm{mean}}(u,c)=\frac{1}{|\mathcal H_u|}\sum_{i\in\mathcal H_u}\langle\mathbf v_i,\mathbf v_c\rangle
\qquad\text{và}\qquad
s_{\mathrm{LSE}}(u,c)=\tau\Big(\log\sum_{i\in\mathcal H_u}\exp\big(\langle\mathbf v_i,\mathbf v_c\rangle/\tau\big)-\log|\mathcal H_u|\Big)
\]
\vspace{-2pt}
\begin{center}
\begin{tikzpicture}[font=\scriptsize,
  pane/.style={rounded corners=2pt,text width=6.3cm,minimum height=1.75cm,align=left,inner sep=5pt,anchor=north west}]
\node[pane,draw=black!45,fill=black!3] (L) at (0,0) {%
  {\bfseries\color{black!75}Mean pooling: tín hiệu bị pha loãng}\par\vspace{3pt}
  Lịch sử: 40 bài thể thao ($\langle\mathbf v_i,\mathbf v_c\rangle\approx0.1$) và 5 bài tài chính ($\approx0.8$).\par
  Ứng viên là bài tài chính: $s_{\mathrm{mean}}=\frac{40\cdot0.1+5\cdot0.8}{45}\approx\mathbf{0.178}$: bị 40 bài thể thao lấn át.};
\node[pane,draw=okc,fill=okc!5,thick] (R) at (6.75,0) {%
  {\bfseries\color{okc}LSE (smooth-max, ví dụ $\tau=0.1$)}\par\vspace{3pt}
  Khuếch đại theo hàm mũ $\exp(\langle\mathbf v_i,\mathbf v_c\rangle/\tau)$ trước khi cộng, nên vài bài rất giống ứng viên chiếm ưu thế.\par
  Cùng ứng viên: $s_{\mathrm{LSE}}\approx\mathbf{0.58}$: giữ được sở thích ngách.};
\end{tikzpicture}
\end{center}
\vspace{-4pt}
\begin{itemize}
  \setlength{\itemsep}{1pt}
  \item Độc giả đọc nhiều chủ đề: độ tương đồng cao với vài bài cùng chủ đề bị chia nhỏ theo hệ số $1/|\mathcal H_u|$.
  \item LSE nằm giữa trung bình ($\tau$ lớn) và MaxSim ($\tau\to0$); $\tau$ thực tế chọn trên validation (số trong hai khung chỉ là ví dụ). Top-k (trung bình $k$ bài giống nhất) có cùng ý tưởng.
\end{itemize}
\ychinh{H8 không thêm tham số học nào; chỉ thay cách gộp các độ tương đồng giữa ứng viên và từng bài đã đọc.}
\par\kern0pt% chiều cao khung tính cả phần sâu của hộp cuối
\note{Ví dụ: độc giả đọc 40 bài thể thao và 5 bài tài chính. Với ứng viên tài chính, mean pooling cho điểm thấp vì 40 bài thể thao kéo trung bình xuống. LSE khuếch đại các bài rất giống ứng viên nên giữ được sở thích ngách. Đây là thay đổi không có tham số học.}
\end{frame}

%% ============================================================================ H10 thiết kế
\begin{frame}[shrink]{H10 \textperiodcentered{} Reader: dùng ứng viên làm câu hỏi để đọc lịch sử}
\begin{columns}[T,onlytextwidth]
\column{0.57\textwidth}
\scriptsize
Tầng nhẹ có huấn luyện, đặt trên bảng embedding cố định $\mathbf E$:
\begin{itemize}
  \setlength{\itemsep}{2pt}
  \item \textbf{Reader theo ứng viên} (\code{um\_frozen}, \code{um\_knn\_rag}): chiếu vector qua MLP hai tầng, $\mathbf h_i=f(\mathbf E_{\mathrm{hist},i})$, $\mathbf c=f(\mathbf E_{\mathrm{cand}})$, rồi dùng $\mathbf c$ làm truy vấn:
  \[
  \alpha_i=\mathrm{softmax}_i\!\Big(\tfrac{\langle\mathbf W_c\mathbf c,\mathbf h_i\rangle}{\sqrt d}\Big),\quad
  s_{\mathrm{CA}}(u,c)=\Big\langle\textstyle\sum_i\alpha_i\mathbf h_i,\ \mathbf c\Big\rangle .
  \]
  \item \textbf{Reader nén một vector} (\code{um\_nrms}, \code{um\_ff}): khối NRMS hoặc Fastformer nén cả lịch sử thành một vector $\mathbf u$ trước khi biết ứng viên.
  \item \textbf{Prior} (\code{um\_prior}): chỉ nhìn bài báo; làm đặc trưng cho GBDT.
\end{itemize}
Huấn luyện: softmax trên bài được click và các bài không được click cùng lượt:
\[
\mathcal L=-\log\frac{\exp s(u,c^+)}{\exp s(u,c^+)+\sum_{c^-}\exp s(u,c^-)} .
\]
\column{0.4\textwidth}
\centering
\begin{tikzpicture}[font=\scriptsize,>=Stealth,
  h/.style={draw=keyc,fill=keyc!6,minimum width=0.7cm,minimum height=0.5cm,inner sep=1pt}]
\node[draw=accentc,fill=accentc!6,minimum width=1.6cm,minimum height=0.55cm] (c) at (1.7,2.6) {ứng viên $\mathbf c$};
\foreach \i/\lab/\w in {0/$\mathbf h_1$/0.4,1/$\mathbf h_2$/1.6,2/$\mathbf h_3$/0.4,3/$\mathbf h_4$/1.1,4/$\mathbf h_5$/0.4}
  {\node[h] (h\i) at (\i*0.85,0.9) {\lab};
   \draw[->,accentc,line width=\w pt] (c.south)--(h\i.north);}
\node[font=\tiny,text=nac,align=center] at (1.7,0.05) {nét càng đậm: trọng số $\alpha_i$ càng lớn\\(bài cùng chủ đề với ứng viên)};
\end{tikzpicture}
\end{columns}

\begin{frame}[shrink]{Validation có trọng số theo tỷ lệ bài mới}
\footnotesize

\[
w_i=\frac{\tau}{\bar c_{\mathrm{val}}}\,c_i+\frac{1-\tau}{1-\bar c_{\mathrm{val}}}\,(1-c_i).
\]

\begin{frame}[shrink]{Rò rỉ trong ngày: vì sao chia fold theo người dùng là sai}
\scriptsize
GBDT được huấn luyện trên validation (150.000 impression $\approx$ 6.05 triệu dòng ứng viên). Chia 4 fold theo \textbf{người dùng} gây quá khớp vào chính ngày đó:
\begin{itemize}
  \setlength{\itemsep}{2pt}
  \item \textbf{Cơ chế:} các impression cùng thời điểm rơi vào cả phần huấn luyện và phần giữ lại của mỗi fold. Cây sâu ($d=7$) dùng các cột độ phân giải cao (tuổi bài, số lần hiển thị, kích thước slate)
  để nhận ra từng bài và nhớ tỷ lệ click của bài đó trong ngày.
  \item \textbf{Dấu hiệu:} điểm out-of-fold cao giả tạo \textbf{0.4922}, nhưng chỉ \textbf{0.3738} trên các bài chưa xuất hiện trong phần huấn luyện.
\end{itemize}
\begin{block}{Cách sửa: cross-validation theo khối thời gian}
\scriptsize
\begin{enumerate}
  \setlength{\itemsep}{1pt}
  \item Chia ngày validation thành \textbf{4 khối thời gian liên tiếp}; bỏ một vùng đệm (tối đa 2\% độ dài ngày, khoảng 29 phút) ở hai biên của khối giữ lại.
  \item Làm thô các cột có thể định danh bài; cấu hình \code{clean} bỏ thêm các cột hằng trong một impression.
  \item Dừng sớm và chọn cấu hình trên \textbf{forward holdout} (60\% đầu ngày để huấn luyện, 10\% để dừng sớm, 30\% cuối để đánh giá), tối ưu trên \textbf{click vào bài mới}.
  Cấu hình được chọn là trung bình của \code{v15} và \code{clean}.
\end{enumerate}
\end{block}

\section{Phụ lục}
%% ============================================================================ Thuật ngữ
\begin{frame}[shrink]{Phụ lục \textperiodcentered{} Thuật ngữ dùng trong bài}
\begin{columns}[T,onlytextwidth]
\column{0.49\textwidth}
\scriptsize
\begin{description}
  \setlength{\itemsep}{1pt}
  \item[Impression] một lượt hiển thị: lịch sử đọc + danh sách ứng viên + nhãn click.
  \item[Bài mới / bài cũ] bài chưa / đã xuất hiện trong dữ liệu huấn luyện (cold / warm).
  \item[Mean pooling] vector người dùng là trung bình vector các bài đã đọc.
  \item[LSE, top-k] gộp độ tương đồng theo kiểu smooth-max hoặc trung bình $k$ bài giống nhất (H8).
  \item[Reader] tầng nhẹ có huấn luyện đọc lịch sử trên embedding cố định; ``theo ứng viên'' nghĩa là ứng viên làm truy vấn attention (H10).
  \item[kNN RAG] cộng vào vector bài một phần vector của các bài gần nhất (H3).
  \item[Vector nối] ghép vector Qwen3 và BGE-large thành 2048 chiều.
  \item[Zero-shot] không huấn luyện thêm gì trên dữ liệu MIND.
\end{description}
\column{0.48\textwidth}
\scriptsize
\begin{description}
  \setlength{\itemsep}{1pt}
  \item[Hợp nhất tuyến tính] cộng có trọng số các điểm đã chuẩn hóa (\code{super\_*}).
  \item[GBDT stacker] cây quyết định tăng cường (XGBoost) học cách kết hợp các điểm và đặc trưng.
  \item[Exposure] thông tin hiển thị của bài lấy từ log, không dùng nhãn: tuổi bài, số lần hiển thị.
  \item[Online / offline] có / không dùng click của chính ngày test.
  \item[Time-block CV] chia fold theo khối thời gian liên tiếp thay vì theo người dùng.
  \item[CI ghép cặp] khoảng tin cậy của hiệu hai mô hình trên cùng các impression.
  \item[Bonferroni] hiệu chỉnh khi so sánh nhiều họ cùng lúc.
  \item[Nhiễu seed] chênh lệch giữa hai lần huấn luyện cùng cấu hình, khác seed.
\end{description}
\end{columns}

\begin{frame}[shrink]{Phụ lục \textperiodcentered{} Câu hỏi có thể gặp (1/2)}
\scriptsize
\begin{description}
  \setlength{\itemsep}{4pt}
  \item[Vì sao không tinh chỉnh LLM end-to-end như PLM-NR?] Đã thử tinh chỉnh một phần theo đúng công thức PLM-NR (H12: mở khóa 2 tầng cuối BGE-large, có nhánh đối chứng đóng băng): +0.0005 (n.s.), Cold AUC $-0.0069$, không qua cổng +0.006 đặt trước. Tinh chỉnh toàn phần vượt khả năng 1 GPU T4; đây là hướng tiếp theo.
  \item[\code{stack\_meta} vượt PLM-NR có công bằng không?] Không cùng điều kiện: \code{stack\_meta} đọc thêm exposure từ log, đo trên \code{MINDlarge\_dev}; số PLM-NR là số công bố trên test benchmark. Phát biểu an toàn: hệ chỉ dùng nội dung + lịch sử (0.4373) còn thấp hơn PLM-NR (0.4438).
  \item[$\tau=0.80$ có rò rỉ thông tin test không?] Có dùng thông tin tổng hợp của test (tỷ lệ click vào bài mới), không dùng nhãn từng mẫu. Đây là giới hạn đã nêu; cách sửa là ước lượng $\tau$ từ tỷ lệ ứng viên mới, không cần nhãn.
  \item[Vì sao \code{stack\_pop} không đem so với các bài báo?] Nó dùng click của chính ngày test theo luồng, thông tin mà mô hình offline không có và không thể nộp lên bảng xếp hạng.
  \item[Semantic ID giảm $-0.1011$ có phải do lỗi?] SID-GPT ngang đối chứng vector ngẫu nhiên (0.2898 so với 0.2897), nên khả năng cao mô hình sinh chưa học được; cần kiểm tra cài đặt trước khi kết luận về phương pháp.
\end{description}

\begin{frame}[shrink]{Phụ lục \textperiodcentered{} Câu hỏi có thể gặp (2/2)}
\scriptsize
\begin{description}
  \setlength{\itemsep}{4pt}
  \item[Khoảng tin cậy có tính nhiễu huấn luyện không?] Không. CI ghép cặp chỉ phản ánh nhiễu mẫu test; nhiễu seed đo riêng (0.007--0.014 trên MIND-small, 0.001--0.007 cho reader trên MIND-large). Chênh nhỏ hơn mức đó giữa hai mô hình đã huấn luyện không được dùng để kết luận.
  \item[Sao không dùng embedding 4B--8B?] Giới hạn GPU. Bằng chứng hiện có cho thấy encoder lớn hơn không tự động tốt hơn (BGE-large chỉ +0.0015 so với BGE-small; Qwen3-0.6B kém BGE-large +0.0102 khi làm đầu vào reader), nhưng chưa loại trừ được.
  \item[Vì sao GBDT không giúp khi chỉ có điểm embedding?] Bốn trọng số tuyến tính đã khai thác hết thông tin của các điểm ($-0.0004$, n.s.); GBDT chỉ có ích khi có thông tin mới như exposure (+0.0147).
  \item[LSE đã so với attention học được, với NAML, Fastformer chưa?] Có. Trên cùng vector cố định: reader kiểu NRMS, Fastformer, 4 truy vấn, theo ứng viên (slide ``H8 so với attention học được''). Baseline đầy đủ NRMS, NAML, Fastformer: slide RQ1. Reader kiểu NAML trên vector cố định chưa chạy.
  \item[Kích thước slate quan trọng nhất có hợp lý không?] Chưa rõ. Bỏ cột này làm điểm holdout giảm; có thể cây dùng nó làm ngữ cảnh để đổi cách kết hợp. Cần ablation riêng.
  \item[Đóng góp chính của đề tài là gì?] (1) Đánh giá chặt 12 họ LLM 2024--2026 cho tin tức với baseline huấn luyện lại, CI ghép cặp và nhiễu seed; (2) chỉ ra điểm nghẽn là tầng tổng hợp lịch sử và hai cách sửa trên embedding cố định (LSE, reader theo ứng viên); (3) giao thức validation có trọng số bài mới + CV theo khối thời gian; (4) hệ ba tầng chạy trên 1 GPU T4.
\end{description}
